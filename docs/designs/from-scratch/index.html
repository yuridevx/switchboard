Договор — это код

Один запрос описан один раз. Им пользуются кнопка, модель, внешний Claude и скрипт.

export const sessionOpen = request("session.open", {
  tool:  "Открыть сессию на машине",
  input: { to: Machine, spec: Spec },
  output:{ callsign: Callsign },
});

// где угодно: страница, Host, служба, скрипт
await call(sessionOpen, { to: "@laptop", spec });
